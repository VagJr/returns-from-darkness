cmake_minimum_required(VERSION 3.24)
project(rfd_th3_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(rfd_room STATIC room_transport.cpp)
target_link_libraries(rfd_room PUBLIC ws2_32 bcrypt)
target_include_directories(rfd_room PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(rfd_th3_room_smoke room_smoke.cpp)
target_link_libraries(rfd_th3_room_smoke PRIVATE rfd_room)
