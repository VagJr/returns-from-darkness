cmake_minimum_required(VERSION 3.24)
project(rfd_peer_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(DISABLE_PACKAGE_CONFIG_AND_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
# libdatachannel's Mbed TLS DTLS implementation declares SRTP profiles even in NO_MEDIA builds.
add_compile_definitions(MBEDTLS_SSL_DTLS_SRTP)
add_subdirectory(../vendor/mbedtls mbedtls)
add_library(MbedTLS::MbedTLS ALIAS mbedtls)
add_subdirectory(../vendor/libdatachannel datachannel)
add_executable(RfdPeer peer_bridge.cpp)
target_include_directories(RfdPeer PRIVATE ../vendor/libdatachannel/deps/json/single_include)
target_link_libraries(RfdPeer PRIVATE datachannel-static winhttp ws2_32 bcrypt)
target_compile_definitions(RfdPeer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
